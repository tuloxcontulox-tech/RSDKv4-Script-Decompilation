// ========================
// Aliases
// ========================

private alias object.value0 : object.timer
private alias object.value1 : object.spinAngle
private alias object.value2 : object.prevRotation

// States
private alias 0 : WRECKINGBALL_INACTIVE
private alias 1 : WRECKINGBALL_ACTIVE
private alias 2 : WRECKINGBALL_EXPLODE


// ========================
// Events
// ========================

event ObjectUpdate
	if object.state == WRECKINGBALL_ACTIVE
		// Animate face every 30 frames (half a second)
		object.timer++
		if object.timer >= 240
			object.timer = 0
		end if

		// Calculate change in swing angle (object.rotation) to rotate the ball realistically
		temp0 = object.rotation
		temp0 -= object.prevRotation

		// Handle wrap-around of object.rotation (which goes from 0 to 511)
		if temp0 > 256
			temp0 -= 512
		else
			if temp0 < -256
				temp0 += 512
			end if
		end if

		// Accumulate change into spin angle
		object.spinAngle += temp0
		object.spinAngle &= 511

		// Store current rotation for next frame
		object.prevRotation = object.rotation
	end if

	if object.state == WRECKINGBALL_EXPLODE
		object.rotation *= 248
		object.rotation >>= 8
		object.timer++
		temp0 = object.timer
		temp0 &= 7
		if temp0 == 4
			Rand(temp0, 32)
			temp0 -= 16
			temp0 <<= 16
			temp0 += object.xpos
			Rand(temp1, 32)
			temp1 -= 16
			temp1 <<= 16
			temp1 += object.ypos
			CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
		end if

		if object.timer == 180
			object.type = TypeName[Blank Object]
		end if
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == WRECKINGBALL_ACTIVE
			BoxCollisionTest(C_TOUCH, object.entityPos, -22, -22, 22, 22, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
		end if
	next
end event


event ObjectDraw
	// Calculate current animated face frame (1 to 8)
	temp0 = object.timer
	temp0 /= 30
	temp0 %= 8
	temp0++

	// Draw the animated face smoothly rotated by our spinAngle
	// We temporarily set object.rotation to spinAngle so DrawSpriteFX uses it!
	temp1 = object.rotation
	object.rotation = object.spinAngle
	DrawSpriteFX(temp0, FX_ROTATE, object.xpos, object.ypos)
	object.rotation = temp1 // Restore original swing angle for any other reference

	// Draw Shine Overlay
	object.inkEffect = INK_ADD
	object.alpha = 160
	DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
end event

	SpriteFrame(-24, -24, 48, 48, 99, 126) 	// Wrecking Ball - #3
	SpriteFrame(-24, -24, 48, 48, 50, 126) 	// Wrecking Ball - #4
	SpriteFrame(-24, -24, 48, 48, 1, 126) 	// Wrecking Ball - #5
	SpriteFrame(-24, -24, 48, 48, 197, 77) 	// Wrecking Ball - #6
	SpriteFrame(-24, -24, 48, 48, 148, 77) 	// Wrecking Ball - #7
