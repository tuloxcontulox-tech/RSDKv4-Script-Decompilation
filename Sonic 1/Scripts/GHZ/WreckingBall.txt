private alias object.value0 : object.timer
private alias object.value10 : object.spinAngle
private alias object.value11 : object.prevRotation

// States
private alias 0 : WRECKINGBALL_INACTIVE
private alias 1 : WRECKINGBALL_ACTIVE
private alias 2 : WRECKINGBALL_EXPLODE


// ========================
// Events
// ========================

event ObjectUpdate
	if object.state == WRECKINGBALL_ACTIVE
		object.timer++
		if object.timer >= 240
			object.timer = 0
		end if
	end if

	if object.state == WRECKINGBALL_EXPLODE
		object.rotation *= 248
		object.rotation >>= 8
		object.timer++
		temp0 = object.timer
		temp0 &= 7
		if temp0 == 4
			Rand(temp0, 32)
			temp0 -= 16
			temp0 <<= 16
			temp0 += object.xpos
			Rand(temp1, 32)
			temp1 -= 16
			temp1 <<= 16
			temp1 += object.ypos
			CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
		end if

		if object.timer == 180
			object.type = TypeName[Blank Object]
		end if
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == WRECKINGBALL_ACTIVE
			BoxCollisionTest(C_TOUCH, object.entityPos, -22, -22, 22, 22, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
		end if

		// Check if player got hurt to trigger laughing face!
		if player[currentPlayer].animation == ANI_HURT
			object.value9 = 120 // laugh for 2 seconds (120 frames)
		end if
	next

	if object.value9 > 0
		object.value9--
	end if

	// Wrecking Ball Trail position tracker
	object.value20++
	if object.value20 >= 2
		object.value20 = 0
		object.value12 = object.value14
		object.value13 = object.value15
		object.value14 = object.value16
		object.value15 = object.value17
		object.value16 = object.value18
		object.value17 = object.value19
		object.value18 = object.xpos
		object.value19 = object.ypos
	end if
end event


event ObjectDraw
	// If the boss was hit recently, use one of the laughing faces (frames 3 to 8).
	// Otherwise, cycle through normal faces (frames 1 and 2).
	if object.value9 > 0
		temp0 = object.timer
		temp0 /= 15 // Animate slightly faster when laughing!
		temp0 %= 6
		temp0 += 3
	else
		temp0 = object.timer
		temp0 /= 30
		temp0 %= 2
		temp0++
	end if

	temp1 = object.rotation
	if object.state == WRECKINGBALL_ACTIVE
		object.spinAngle = object.rotation
		object.spinAngle -= 128
		object.spinAngle &= 511
	else
		object.spinAngle = 0
	end if

	// Dibujar trail con rotacion combinada con tinta
	if object.value12 != 0
		temp5 = object.rotation
		object.rotation = object.spinAngle
		object.inkEffect = INK_ALPHA
		temp2 = FX_ROTATE
		temp2 |= FX_INK

		// Copia 4 (Mas lejana / mas transparente)
		object.alpha = 40
		DrawSpriteFX(temp0, temp2, object.value12, object.value13)

		// Copia 3
		object.alpha = 80
		DrawSpriteFX(temp0, temp2, object.value14, object.value15)

		// Copia 2
		object.alpha = 120
		DrawSpriteFX(temp0, temp2, object.value16, object.value17)

		// Copia 1 (Mas cercana / mas visible)
		object.alpha = 160
		DrawSpriteFX(temp0, temp2, object.value18, object.value19)

		object.inkEffect = INK_NONE
		object.rotation = temp5
	end if

	// Dibujar la bola principal
	object.rotation = object.spinAngle
	DrawSpriteFX(temp0, FX_ROTATE, object.xpos, object.ypos)
	object.rotation = temp1
end event

	SpriteFrame(-24, -24, 48, 48, 99, 126) 	// Wrecking Ball - #3
	SpriteFrame(-24, -24, 48, 48, 50, 126) 	// Wrecking Ball - #4
	SpriteFrame(-24, -24, 48, 48, 1, 126) 	// Wrecking Ball - #5
	SpriteFrame(-24, -24, 48, 48, 197, 77) 	// Wrecking Ball - #6
	SpriteFrame(-24, -24, 48, 48, 148, 77) 	// Wrecking Ball - #7
	SpriteFrame(-24, -24, 48, 48, 99, 77) 	// Wrecking Ball - #8
end event
