private function Sega_CheckTouchCheatCodes
	CheckTouchRect(0, 0, screen.xsize, screen.ysize)
	if checkResult == -1
		sega.touchDown = false
	end if

	if sega.touchDown == false
		temp0 = screen.xcenter
		switch sega.levelSelectCheatPos
		case 0
			temp0 -= 93
			temp1 = temp0
			temp1 += 39
			break
			
		case 1
			temp0 -= 50
			temp1 = temp0
			temp1 += 42
			break

		case 2
			temp0 -= 6
			temp1 = temp0
			temp1 += 42
			break

		case 3
			temp0 += 36
			temp1 = temp0
			temp1 += 58
			break

		end switch

		CheckTouchRect(temp0, 90, temp1, 150)
		if checkResult > -1
			sega.touchDown = true
			sega.levelSelectCheatPos++
			if sega.levelSelectCheatPos == 4
				PlaySfx(SfxName[Ring L], false)
				options.stageSelectFlag = true
			end if
		end if

		temp0 = screen.xcenter
		switch sega.regionSwitchCheatPos
		case 0
			temp0 += 36
			temp1 = temp0
			temp1 += 58
			break

		case 1
			temp0 -= 6
			temp1 = temp0
			temp1 += 42
			break

		case 2
			temp0 -= 50
			temp1 = temp0
			temp1 += 42
			break

		case 3
			temp0 -= 93
			temp1 = temp0
			temp1 += 39
			break

		end switch

		CheckTouchRect(temp0, 90, temp1, 150)
		if checkResult > -1
			sega.touchDown = true
			sega.regionSwitchCheatPos++
			if sega.regionSwitchCheatPos == 4
				PlaySfx(SfxName[Ring L], false)
				options.region ^= true
			end if
		end if
	end if
end function

event ObjectUpdate
#platform: USE_ORIGINS
	if game.titleMode != 0
		game.titleMode = 0
		ResetObjectEntity(object.entityPos, TypeName[Logo], 0, 0, 0)
	else
#endplatform
		switch object.state
		case SEGA_SETUP
			object.alpha = 0x00
			object.inkEffect = INK_ALPHA
			sega.timer = 120
			sega.sonicX = -40
			sega.animFrame = 0
			SetScreenFade(0, 0, 0, sega.timer)
			object.state++
			break

		case SEGA_WALK_INTRO
			if sega.timer > 0
				sega.timer -= 4
			end if
			SetScreenFade(0, 0, 0, sega.timer)

			sega.sonicX += 2
			sega.animFrame++
			if sega.animFrame >= 24
				sega.animFrame = 0
			end if

			if sega.sonicX >= screen.xcenter
				sega.animFrame = 0
				object.state = SEGA_RUN_OUT
			end if
			break

		case SEGA_RUN_OUT
			sega.sonicX += 6
			sega.animFrame++
			if sega.animFrame >= 16
				sega.animFrame = 0
			end if

			temp0 = screen.xsize
			temp0 += 40
			if sega.sonicX > temp0
				sega.timer = 0
				sega.animFrame = 0
				PlaySfx(SfxName[Sega Chant], false)
				object.state = SEGA_3D_WAVE
			end if
			break

		case SEGA_3D_WAVE
			sega.timer++
			sega.animFrame++
			if sega.animFrame >= 20
				sega.animFrame = 0
			end if

			if sega.timer == 120
				sega.timer = 0
				sega.flashTimer = 30
				object.state = SEGA_FLASH_HORROR
			end if
#platform: USE_STANDALONE
			CallFunction(Sega_CheckTouchCheatCodes)
#endplatform
			break

		case SEGA_FLASH_HORROR
			if sega.flashTimer > 0
				sega.flashTimer--
			else
				sega.timer = 0
				object.state = SEGA_RED_FLICKER
			end if
			break

		case SEGA_RED_FLICKER
			sega.timer++
			if sega.timer >= 120
				sega.timer = 0
				sega.animFrame = 0
				object.state = SEGA_LETRAS_DISPERSE
			end if
			break

		case SEGA_LETRAS_DISPERSE
			sega.timer++
			sega.animFrame++
			if sega.timer >= 90
				sega.timer = 0
				object.state = SEGA_FADEOUT
			end if
			break

		case SEGA_FADEOUT
			if sega.timer < 256
				sega.timer += 8
				SetScreenFade(0, 0, 0, sega.timer)
			else
				ResetObjectEntity(object.entityPos, TypeName[Sonic Team], 0, object.xpos, object.ypos)
				SetScreenFade(0, 0, 0, 0xFF)
			end if
			break
			
		end switch
#platform: USE_ORIGINS
	end if // game.titleMode == 0 check from way up above
#endplatform
end event


event ObjectDraw
	switch object.state
	case SEGA_SETUP
	case SEGA_WALK_INTRO
	case SEGA_RUN_OUT
		// Custom Blue Background (RGB 19, 66, 148)
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 0xFF)

		// Draw White Walking / Running Sonic
		if object.state == SEGA_WALK_INTRO
			temp0 = sega.animFrame
			temp0 /= 6
			temp0 += 2 // Walking frames (frames #2-#5)
			DrawSpriteScreenXY(temp0, sega.sonicX, screen.ycenter)
		else
			temp0 = sega.animFrame
			temp0 /= 4
			temp0 += 6 // Running frames (frames #6-#9)
			DrawSpriteScreenXY(temp0, sega.sonicX, screen.ycenter)
		end if
		break

	case SEGA_3D_WAVE
		// Blue Background
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 0xFF)

		// 3D Wave Sega Logo
		temp0 = sega.animFrame
		temp0 /= 5
		temp0 += 10 // 3D Wave SEGA Logo frames (#10-#13)
		DrawSpriteScreenXY(temp0, screen.xcenter, screen.ycenter)
		break

	case SEGA_FLASH_HORROR
		// White Flash
		DrawRect(0, 0, screen.xsize, screen.ysize, 255, 255, 255, 0xFF)
		break

	case SEGA_RED_FLICKER
		// Creepy Forest Background (Forest.gif - Frame #18)
		DrawSpriteScreenXY(18, screen.xcenter, screen.ycenter)

		// Red SEGA flickering (#14 / #15)
		temp0 = sega.timer
		temp0 &= 4
		if temp0 == 0
			DrawSpriteScreenXY(14, screen.xcenter, screen.ycenter)
		else
			DrawSpriteScreenXY(15, screen.xcenter, screen.ycenter)
		end if
		break

	case SEGA_LETRAS_DISPERSE
		// Creepy Forest Background (Forest.gif - Frame #18)
		DrawSpriteScreenXY(18, screen.xcenter, screen.ycenter)

		// Outlined / Dispersing SEGA letters (#16 - together, #17 - separated/dispersing)
		if sega.animFrame < 30
			DrawSpriteScreenXY(16, screen.xcenter, screen.ycenter)
		else
			DrawSpriteScreenXY(17, screen.xcenter, screen.ycenter)
		end if
		break

	case SEGA_FADEOUT
		// Creepy Forest Background during smooth fadeout
		DrawSpriteScreenXY(18, screen.xcenter, screen.ycenter)
		DrawSpriteScreenXY(17, screen.xcenter, screen.ycenter)
		break

	end switch
end event
