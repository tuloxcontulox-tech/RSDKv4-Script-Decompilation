event ObjectUpdate
#platform: USE_ORIGINS
	if game.titleMode != 0
		game.titleMode = 0
		ResetObjectEntity(object.entityPos, TypeName[Logo], 0, 0, 0)
	else
#endplatform
		switch object.state
		case SEGA_SETUP
			object.alpha = 0x00
			object.inkEffect = INK_ALPHA
			sega.timer = 0
			object.state = SEGA_SEGASCREEN
			break

		case SEGA_SEGASCREEN
			sega.timer++
			if sega.timer == 360
				sega.timer = 0
				object.state = SEGA_FADEOUT
			end if

			if sega.timer == 2
				PlaySfx(SfxName[Sega Chant], false)
			end if
			break

		case SEGA_FADEOUT
			if sega.timer < 256
				sega.timer += 8
				SetScreenFade(0, 0, 0, sega.timer)
			else
				ResetObjectEntity(object.entityPos, TypeName[Sonic Team], 0, 0, 0)
				SetScreenFade(0, 0, 0, 0xFF)
			end if
			break
			
		end switch
#platform: USE_ORIGINS
	end if
#endplatform
end event


event ObjectDraw
	// Draw Solid Black Background
	DrawRect(0, 0, screen.xsize, screen.ysize, 0x00, 0x00, 0x00, 0xFF)

	// Draw DEL
	if sega.timer >= 27
		if sega.timer < 57
			temp0 = sega.timer
			temp0 -= 27
			temp0 *= 8
			object.alpha = temp0
		else
			object.alpha = 255
		end if
		temp1 = screen.xcenter
		temp1 -= 80
		DrawSpriteScreenFX(0, FX_INK, temp1, screen.ycenter)
	end if

	// Draw TA
	if sega.timer >= 60
		if sega.timer < 90
			temp0 = sega.timer
			temp0 -= 60
			temp0 *= 8
			object.alpha = temp0
		else
			object.alpha = 255
		end if
		temp1 = screen.xcenter
		temp1 -= 18
		DrawSpriteScreenFX(1, FX_INK, temp1, screen.ycenter)
	end if

	// Draw RUNE
	if sega.timer >= 120
		if sega.timer < 150
			temp0 = sega.timer
			temp0 -= 120
			temp0 *= 8
			object.alpha = temp0
		else
			object.alpha = 255
		end if
		temp1 = screen.xcenter
		temp1 += 62
		DrawSpriteScreenFX(2, FX_INK, temp1, screen.ycenter)
	end if
end event
