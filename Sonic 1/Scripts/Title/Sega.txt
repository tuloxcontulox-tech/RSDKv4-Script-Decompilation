private function Sega_CheckTouchCheatCodes
	CheckTouchRect(0, 0, screen.xsize, screen.ysize)
	if checkResult == -1
		sega.touchDown = false
	end if

	if sega.touchDown == false
		temp0 = screen.xcenter
		switch sega.levelSelectCheatPos
		case 0
			temp0 -= 93
			temp1 = temp0
			temp1 += 39
			break
			
		case 1
			temp0 -= 50
			temp1 = temp0
			temp1 += 42
			break

		case 2
			temp0 -= 6
			temp1 = temp0
			temp1 += 42
			break

		case 3
			temp0 += 36
			temp1 = temp0
			temp1 += 58
			break

		end switch

		CheckTouchRect(temp0, 90, temp1, 150)
		if checkResult > -1
			sega.touchDown = true
			sega.levelSelectCheatPos++
			if sega.levelSelectCheatPos == 4
				PlaySfx(SfxName[Ring L], false)
				options.stageSelectFlag = true
			end if
		end if

		temp0 = screen.xcenter
		switch sega.regionSwitchCheatPos
		case 0
			temp0 += 36
			temp1 = temp0
			temp1 += 58
			break

		case 1
			temp0 -= 6
			temp1 = temp0
			temp1 += 42
			break

		case 2
			temp0 -= 50
			temp1 = temp0
			temp1 += 42
			break

		case 3
			temp0 -= 93
			temp1 = temp0
			temp1 += 39
			break

		end switch

		CheckTouchRect(temp0, 90, temp1, 150)
		if checkResult > -1
			sega.touchDown = true
			sega.regionSwitchCheatPos++
			if sega.regionSwitchCheatPos == 4
				PlaySfx(SfxName[Ring L], false)
				options.region ^= true
			end if
		end if
	end if
end function

event ObjectUpdate
#platform: USE_ORIGINS
	if game.titleMode != 0
		game.titleMode = 0
		ResetObjectEntity(object.entityPos, TypeName[Sonic Team], 0, 0, 0)
	else
#endplatform
		switch object.state
		case SEGA_WALK_INTRO
			if sega.timer == 0
				object.alpha = 0xFF
				object.inkEffect = INK_NONE
				sega.sonicX = -20
				sega.animFrame = 0
				sega.scale = 512
				sega.rotation = 180
				sega.waveAngle = 0
				PlaySfx(SfxName[Sega Chant], false)
				SetScreenFade(0, 0, 0, 0)
			end if

			// Walk for 1.70s (102 frames at 60 FPS)
			sega.timer++
			sega.sonicX += 1
			sega.animFrame++
			if sega.animFrame >= 24
				sega.animFrame = 0
			end if

			if sega.timer >= 102
				sega.timer = 0
				sega.animFrame = 0
				object.state = SEGA_RUN_OUT
			end if
			break

		case SEGA_RUN_OUT
			// Run for 1.70s (102 frames at 60 FPS) and accelerate off screen
			sega.timer++
			sega.sonicX += 5
			sega.animFrame++
			if sega.animFrame >= 16
				sega.animFrame = 0
			end if

			if sega.timer >= 102
				sega.timer = 0
				sega.animFrame = 0
				object.state = SEGA_3D_WAVE_ENTRY
			end if
			break

		case SEGA_3D_WAVE_ENTRY
			// 3D SEGA logo entrance: scale down from large to centered, rotate 180 -> 0, apply sine wave
			sega.timer++
			sega.animFrame++
			if sega.animFrame >= 20
				sega.animFrame = 0
			end if

			if sega.scale > 256
				sega.scale -= 4
			else
				sega.scale = 256
			end if

			if sega.rotation > 0
				sega.rotation -= 3
			else
				sega.rotation = 0
			end if

			sega.waveAngle += 8
			sega.waveAngle &= 0xFF

			if sega.timer >= 120
				sega.timer = 0
				sega.flashTimer = 3
				object.state = SEGA_FLASH_FOREST
			end if
#platform: USE_STANDALONE
			CallFunction(Sega_CheckTouchCheatCodes)
#endplatform
			break

		case SEGA_FLASH_FOREST
			// Pure white flash for 3 frames
			if sega.flashTimer > 0
				sega.flashTimer--
			else
				sega.timer = 0
				object.state = SEGA_RED_FLICKER_ZOOM
			end if
			break

		case SEGA_RED_FLICKER_ZOOM
			// Rapid flickering (1-2 frames), then separated letters spin and zoom in towards screen with fadeout
			sega.timer++
			if sega.timer > 30
				sega.scale += 12
				sega.rotation += 16
				if sega.timer < 120
					temp0 = sega.timer
					temp0 -= 30
					temp0 *= 3
					SetScreenFade(0, 0, 0, temp0)
				end if
			end if

			if sega.timer >= 120
				sega.timer = 0
				object.state = SEGA_END_INTRO
			end if
			break

		case SEGA_END_INTRO
			if sega.timer < 256
				sega.timer += 16
				SetScreenFade(0, 0, 0, sega.timer)
			else
				ResetObjectEntity(object.entityPos, TypeName[Sonic Team], 0, object.xpos, object.ypos)
				SetScreenFade(0, 0, 0, 0xFF)
			end if
			break
			
		end switch
#platform: USE_ORIGINS
	end if // game.titleMode == 0 check from way up above
#endplatform
end event


event ObjectDraw
	switch object.state
	case SEGA_WALK_INTRO
	case SEGA_RUN_OUT
		// Custom Blue Background (RGB 19, 66, 148) for 320x240 screen
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 0xFF)

		// Draw White Walking / Running Sonic
		if object.state == SEGA_WALK_INTRO
			temp0 = sega.animFrame
			temp0 /= 6
			temp0 += 2 // Walking frames (#2-#5)
			DrawSpriteScreenXY(temp0, sega.sonicX, 120)
		else
			temp0 = sega.animFrame
			temp0 /= 4
			temp0 += 6 // Running frames (#6-#9)
			DrawSpriteScreenXY(temp0, sega.sonicX, 120)
		end if
		break

	case SEGA_3D_WAVE_ENTRY
		// Blue Background
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 0xFF)

		// Calculate sine wave displacement
		Sin(temp1, sega.waveAngle)
		temp1 >>= 5
		temp1 += 120

		// 3D Wave Sega Logo with rotation, scale, and sine wave offset
		temp0 = sega.animFrame
		temp0 /= 5
		temp0 += 10 // 3D Wave SEGA Logo frames (#10-#13)
		object.scale = sega.scale
		object.rotation = sega.rotation
		DrawSpriteScreenFX(temp0, FX_ROTSCALE, screen.xcenter, temp1)
		break

	case SEGA_FLASH_FOREST
		// Pure White Flash
		DrawRect(0, 0, screen.xsize, screen.ysize, 255, 255, 255, 0xFF)
		break

	case SEGA_RED_FLICKER_ZOOM
		// 320x240 Creepy Forest Background (Forest.gif - Frame #21)
		DrawSpriteScreenXY(21, screen.xcenter, 120)

		if sega.timer <= 30
			// Rapid red SEGA logo flickering (#14 / #15)
			temp0 = sega.timer
			temp0 &= 2
			if temp0 == 0
				DrawSpriteScreenXY(14, screen.xcenter, 120)
			else
				DrawSpriteScreenXY(15, screen.xcenter, 120)
			end if
		else
			// Separated individual SEGA letters (S, E, G, A) spinning and zooming in towards camera
			object.scale = sega.scale
			object.rotation = sega.rotation

			// Letter 'S' (#17)
			temp0 = screen.xcenter
			temp0 -= 48
			DrawSpriteScreenFX(17, FX_ROTSCALE, temp0, 120)

			// Letter 'E' (#18)
			temp0 = screen.xcenter
			temp0 -= 16
			DrawSpriteScreenFX(18, FX_ROTSCALE, temp0, 120)

			// Letter 'G' (#19)
			temp0 = screen.xcenter
			temp0 += 16
			DrawSpriteScreenFX(19, FX_ROTSCALE, temp0, 120)

			// Letter 'A' (#20)
			temp0 = screen.xcenter
			temp0 += 48
			DrawSpriteScreenFX(20, FX_ROTSCALE, temp0, 120)
		end if
		break

	case SEGA_END_INTRO
		// Forest background during final dark fadeout
		DrawSpriteScreenXY(21, screen.xcenter, 120)
		break

	end switch
end event
