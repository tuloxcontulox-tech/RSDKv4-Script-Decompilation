// Aliases

private alias object.value0  : object.timer
private alias object.value1  : sonic.xpos
private alias object.value2  : sonic.ypos
private alias object.value3  : sonic.frame
private alias object.value4  : sonic.animTimer
private alias object.value5  : sonic.speed
private alias object.value6  : sega.scale
private alias object.value7  : sega.rotation
private alias object.value8  : sega.waveTimer
private alias object.value9  : flash.timer
private alias object.value10 : blink.timer
private alias object.value11 : fade.alpha
private alias object.value12 : letter.scale
private alias object.value13 : letter.rotation

// States
private alias 0 : STATE_WALKING
private alias 1 : STATE_RUNNING
private alias 2 : STATE_SEGA_WHITE_ENTRANCE
private alias 3 : STATE_FLASH_TRANSITION
private alias 4 : STATE_LETTERS_ZOOM
private alias 5 : STATE_END_INTRO

// Ink Effects
public alias 0 : INK_NONE
public alias 1 : INK_BLEND
public alias 2 : INK_ALPHA
public alias 3 : INK_ADD
public alias 4 : INK_SUB

// FX Modes
public alias 0 : FX_NONE
public alias 1 : FX_FLIP
public alias 2 : FX_ROTATE
public alias 3 : FX_ROTOSCALE
public alias 4 : FX_SCALE

// Priority
public alias 0 : PRIORITY_BOUNDS
public alias 1 : PRIORITY_ACTIVE
public alias 2 : PRIORITY_ALWAYS


// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case STATE_WALKING
		if object.timer == 0
			PlaySfx(SfxName[Sega], false)
			sonic.xpos = -30
			sonic.ypos = screen.ycenter
			sonic.ypos += 20
			sonic.frame = 0
			sonic.animTimer = 0
			sonic.speed = 2
		end if

		sonic.xpos += sonic.speed

		sonic.animTimer++
		if sonic.animTimer >= 8
			sonic.animTimer = 0
			sonic.frame++
			if sonic.frame >= 6
				sonic.frame = 0
			end if
		end if

		object.timer++
		if object.timer >= 102
			object.timer = 0
			sonic.frame = 6
			sonic.animTimer = 0
			sonic.speed = 5
			object.state = STATE_RUNNING
		end if
		break

	case STATE_RUNNING
		sonic.xpos += sonic.speed

		sonic.animTimer++
		if sonic.animTimer >= 4
			sonic.animTimer = 0
			sonic.frame++
			if sonic.frame >= 8
				sonic.frame = 6
			end if
		end if

		object.timer++
		if object.timer >= 102
			object.timer = 0
			sega.scale = 0x400
			sega.rotation = 256
			sega.waveTimer = 0
			object.state = STATE_SEGA_WHITE_ENTRANCE
		end if
		break

	case STATE_SEGA_WHITE_ENTRANCE
		object.timer++
		sega.waveTimer += 8

		if sega.scale > 0x200
			sega.scale -= 8
		else
			sega.scale = 0x200
		end if

		if sega.rotation > 0
			sega.rotation -= 4
		else
			sega.rotation = 0
		end if

		if object.timer >= 120
			object.timer = 0
			flash.timer = 0
			object.state = STATE_FLASH_TRANSITION
		end if
		break

	case STATE_FLASH_TRANSITION
		flash.timer++
		if flash.timer >= 3
			object.timer = 0
			blink.timer = 0
			fade.alpha = 0
			letter.scale = 0x200
			letter.rotation = 0
			object.state = STATE_LETTERS_ZOOM
		end if
		break

	case STATE_LETTERS_ZOOM
		object.timer++
		blink.timer++

		if object.timer > 10
			letter.scale += 0x20
			letter.rotation += 8

			if fade.alpha < 255
				fade.alpha += 4
				if fade.alpha > 255
					fade.alpha = 255
				end if
			end if
		end if

		if object.timer >= 90
			object.state = STATE_END_INTRO
		end if
		break

	case STATE_END_INTRO
		SetScreenFade(0, 0, 0, 255)
		ResetObjectEntity(object.entityPos, TypeName[Logo], 0, object.xpos, object.ypos)
		break

	end switch
end event


event ObjectDraw
	switch object.state
	case STATE_WALKING
	case STATE_RUNNING
		// Draw Solid Blue Background (RGB: 19, 66, 148)
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 255)

		// Draw Sonic
		DrawSpriteScreenXY(sonic.frame, sonic.xpos, sonic.ypos)
		break

	case STATE_SEGA_WHITE_ENTRANCE
		// Draw Solid Blue Background
		DrawRect(0, 0, screen.xsize, screen.ysize, 19, 66, 148, 255)

		// Draw 3D White SEGA Intro Logo with wave & scale/rotation FX
		object.scale = sega.scale
		object.rotation = sega.rotation
		object.inkEffect = INK_NONE

		// Apply wave offset calculation
		Cos(temp0, sega.waveTimer)
		temp0 >>= 5
		temp1 = screen.ycenter
		temp1 += temp0

		if object.rotation != 0
			DrawSpriteScreenFX(8, FX_ROTOSCALE, screen.xcenter, temp1)
		else
			DrawSpriteScreenFX(8, FX_SCALE, screen.xcenter, temp1)
		end if
		break

	case STATE_FLASH_TRANSITION
		// Render Forest Background (Frame 14)
		DrawSpriteScreenXY(14, screen.xcenter, screen.ycenter)

		// Render SEGA ROJO Logo (Frame 9)
		DrawSpriteScreenXY(9, screen.xcenter, screen.ycenter)

		// Draw Pure White Flash (RGB: 255, 255, 255) for 2-3 frames
		DrawRect(0, 0, screen.xsize, screen.ysize, 255, 255, 255, 255)
		break

	case STATE_LETTERS_ZOOM
		// Render Forest Background
		DrawSpriteScreenXY(14, screen.xcenter, screen.ycenter)

		temp0 = blink.timer
		temp0 &= 2
		if temp0 == 0
			// Draw SEGA ROJO Logo during initial rapid blink
			DrawSpriteScreenXY(9, screen.xcenter, screen.ycenter)
		else
			// Draw Separated SEGA 3D Letters (S, E, G, A) with independent scale and rotation
			object.scale = letter.scale
			object.rotation = letter.rotation
			object.inkEffect = INK_NONE

			// Letter 'S' (Frame 10)
			temp2 = screen.xcenter
			temp2 -= 60
			DrawSpriteScreenFX(10, FX_ROTOSCALE, temp2, screen.ycenter)

			// Letter 'E' (Frame 11)
			temp2 = screen.xcenter
			temp2 -= 20
			DrawSpriteScreenFX(11, FX_ROTOSCALE, temp2, screen.ycenter)

			// Letter 'G' (Frame 12)
			temp2 = screen.xcenter
			temp2 += 20
			DrawSpriteScreenFX(12, FX_ROTOSCALE, temp2, screen.ycenter)

			// Letter 'A' (Frame 13)
			temp2 = screen.xcenter
			temp2 += 60
			DrawSpriteScreenFX(13, FX_ROTOSCALE, temp2, screen.ycenter)
		end if

		// Apply Fade Out darkening screen overlay
		if fade.alpha > 0
			DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, fade.alpha)
		end if
		break

	case STATE_END_INTRO
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		break

	end switch
end event
