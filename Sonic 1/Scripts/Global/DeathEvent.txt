// ========================
// Aliases
// ========================

private alias object.value1 : object.leftTextPos
private alias object.value2 : object.rightTextPos
private alias object.value3 : object.timer
private alias object.value4 : object.timeOver

// States
private alias 0 : DEATHEVENT_GAMEOVER
private alias 1 : DEATHEVENT_TIMEOVER
private alias 2 : DEATHEVENT_DEATH
private alias 3 : DEATHEVENT_DEATH_TA
private alias 4 : DEATHEVENT_WAITFORCALLBACK
private alias 5 : DEATHEVENT_RETRY
private alias 6 : DEATHEVENT_SETSCREENFADE

// Game Modes Aliases
private alias 1 : MODE_SAVEGAME

// Presentation Stage Aliases
private alias 0 : STAGE_P_TITLE
private alias 4 : STAGE_P_CONTINUE


// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case DEATHEVENT_GAMEOVER
		if object.leftTextPos != 0
			StopMusic()
			SetMusicTrack("man.ogg", 5,  0) // Load man.ogg into track 5 with loop!
			PlayMusic(5)
			object.leftTextPos = 0
			object.rightTextPos = 0 // Reset rightTextPos to 0 so the countdown does not run automatically!
			object.timer = 0
			object.visible = true // Force visible to draw the tree
			object[9].type = TypeName[Blank Object] // Hide HUD completely
			object[1].visible = false // Hide player 2

			// Load custom palette once during initialization
			LoadPalette("Man", 0, 0, 0, 255)
			end if

		// Overwrite the first few colors of the active palette as a safeguard for correct tree colors
		SetPaletteEntry(0, 128, 0x000000)
		SetPaletteEntry(0, 129, 0x0B0B32)
		SetPaletteEntry(0, 130, 0x202040)
		SetPaletteEntry(0, 131, 0x612C61)
		SetPaletteEntry(0, 132, 0xA00080)
		SetPaletteEntry(0, 133, 0xC00080)
		SetPaletteEntry(0, 134, 0xE02040)
		SetPaletteEntry(0, 135, 0xFF8080)

		// Fade-in the black background and centered tree in half a second
		if object.timer < 255
			object.timer += 9
			if object.timer > 255
				object.timer = 255
			end if
		end if

		// Animate the tree parts every 0.25s (15 frames at 60 FPS) across an 8-step cycle
		object.animationTimer++
		temp0 = object.animationTimer
		temp0 /= 15
		temp0 %= 8

		// Purple Foliage offsets (value5 = X, value6 = Y): 2x UP, 2x LEFT, 2x DOWN, 2x RIGHT
		switch temp0
		case 0
			object.value5 = 0
			object.value6 = -1
			break
		case 1
			object.value5 = 0
			object.value6 = -2
			break
		case 2
			object.value5 = -1
			object.value6 = -2
			break
		case 3
			object.value5 = -2
			object.value6 = -2
			break
		case 4
			object.value5 = -2
			object.value6 = -1
			break
		case 5
			object.value5 = -2
			object.value6 = 0
			break
		case 6
			object.value5 = -1
			object.value6 = 0
			break
		case 7
			object.value5 = 0
			object.value6 = 0
			break
		end switch

		// Red Foliage offsets (value7 = X, value8 = Y): 2x DOWN, 2x RIGHT, 2x UP, 2x LEFT
		switch temp0
		case 0
			object.value7 = 0
			object.value8 = 1
			break
		case 1
			object.value7 = 0
			object.value8 = 2
			break
		case 2
			object.value7 = 1
			object.value8 = 2
			break
		case 3
			object.value7 = 2
			object.value8 = 2
			break
		case 4
			object.value7 = 2
			object.value8 = 1
			break
		case 5
			object.value7 = 2
			object.value8 = 0
			break
		case 6
			object.value7 = 1
			object.value8 = 0
			break
		case 7
			object.value7 = 0
			object.value8 = 0
			break
		end switch

		// Wait for interaction button to exit
		temp0 = false
		if keyPress[0].buttonA == true
			temp0 = true
		end if
		if keyPress[0].buttonB == true
			temp0 = true
		end if
		if keyPress[0].buttonC == true
			temp0 = true
		end if
		if keyPress[0].start == true
			temp0 = true
		end if
#platform: MOBILE
		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			temp0 = true
		end if
#endplatform

		if temp0 == true
			PlaySfx(SfxName[Weird], false)
			recStageList = stage.activeList
			recStage = stage.listPos
			stage.activeList = PRESENTATION_STAGE
			stage.listPos = STAGE_P_TITLE
#platform: USE_STANDALONE
			engine.state = 8
#endplatform
			lampPostID = 0
			LoadStage()
		end if
		break

	case DEATHEVENT_TIMEOVER
#platform: USE_ORIGINS
		object.timeOver |= object.state
		
		if game.playMode == BOOT_PLAYMODE_BOSSRUSH
			object.type = TypeName[Blank Object]
		end if
#endplatform

		temp0 = screen.xcenter
		temp0 -= 72
		if object.leftTextPos < temp0
			object.leftTextPos += 16
			if object.leftTextPos > temp0
				object.leftTextPos = temp0
			end if
		end if

		temp0 += 80
		if object.rightTextPos > temp0
			object.rightTextPos -= 16
			if object.rightTextPos < temp0
				object.rightTextPos = temp0
			end if
		end if

		if object.timer < 288
			object.timer += 4
		else
#platform: USE_ORIGINS
			if object.timeOver == true
				if lampPostID > 0
					recMilliSeconds = 99
					recSeconds = 59
					recMinutes = 9
					game.timeOver = true
				else
					recMilliSeconds = 0
					recSeconds = 0
					recMinutes = 0
				end if
			end if
#endplatform

			if object.state == DEATHEVENT_GAMEOVER
				recStageList = stage.activeList
				recStage = stage.listPos
				stage.activeList = PRESENTATION_STAGE

				if player.continues == 0
					stage.listPos = STAGE_P_TITLE
#platform: USE_STANDALONE
					engine.state = 8
#endplatform
					CallNativeFunction2(ShowPromoPopup, 0, "GameOverPromo") // still in Origins btw
				else
					stage.listPos = STAGE_P_CONTINUE
				end if
				
				lampPostID = 0

				if options.gameMode == MODE_SAVEGAME
					arrayPos1 = options.saveSlot
					arrayPos1 <<= 3
					arrayPos1++
					if saveRAM[arrayPos1] < 3 // lives
						saveRAM[arrayPos1] = 3
					end if
					arrayPos1++
					saveRAM[arrayPos1] = 0 // score
					arrayPos1++
					saveRAM[arrayPos1] = 50000 // score bonus
					player.score = 0
					WriteSaveRAM()
				end if
			else
				if options.gameMode == MODE_SAVEGAME
					arrayPos1 = options.saveSlot
					arrayPos1 <<= 3
					arrayPos1++
					saveRAM[arrayPos1] = player.lives
					WriteSaveRAM()
				end if
			end if

#platform: USE_STANDALONE
			if object.state == DEATHEVENT_TIMEOVER
				recMilliSeconds = 0
				recSeconds = 0
				recMinutes = 0
			end if
#endplatform

#platform: USE_ORIGINS
			if object.state == DEATHEVENT_GAMEOVER
				CallNativeFunction2(NotifyCallback, NOTIFY_TIME_OVER, game.timeOver)
			end if
			
			if object.state == DEATHEVENT_TIMEOVER
				player.deadResetRings = true
			end if
#endplatform

			LoadStage()
		end if

		if object.timer > 0
			music.volume -= 2
			SetScreenFade(0, 0, 0, object.timer)
		else
			if keyPress[0].buttonA == true
				object.timer = 0
			end if
			
			if keyPress[0].buttonB == true
				object.timer = 0
			end if
			
			if keyPress[0].buttonC == true
				object.timer = 0
			end if
			
#platform: MOBILE
			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				object.timer = 0
			end if
#endplatform
		end if
		break

	case DEATHEVENT_DEATH
#platform: USE_ORIGINS
		CheckNotEqual(game.playMode, BOOT_PLAYMODE_BOSSRUSH)
		temp0 = checkResult
		CheckNotEqual(game.oneStageFlag, false)
		checkResult &= temp0
		
		if checkResult != false
			game.callbackResult = -1
			CallNativeFunction4(NotifyCallback, NOTIFY_STAGE_RETRY, false, stage.listPos, 0)
			object.state = DEATHEVENT_WAITFORCALLBACK
			if options.gameMode == MODE_SAVEGAME
				arrayPos1 = options.saveSlot
				arrayPos1 <<= 3
				arrayPos1++
				saveRAM[arrayPos1] = player.lives
				WriteSaveRAM()
			end if
		else
#endplatform
			if object.timer < 288
				music.volume -= 2
				object.timer += 4
			else
				if options.gameMode == MODE_SAVEGAME
					arrayPos1 = options.saveSlot
					arrayPos1 <<= 3
					arrayPos1++
					saveRAM[arrayPos1] = player.lives
					WriteSaveRAM()
				end if
				
#platform: USE_STANDALONE
				LoadStage()
#endplatform

#platform: USE_ORIGINS
				if game.playMode != BOOT_PLAYMODE_MISSION
					LoadStage()
				end if
#endplatform
			end if
			
			SetScreenFade(0, 0, 0, object.timer)
#platform: USE_ORIGINS
		end if
		
		player.deadResetRings = true
#endplatform
		break

	case DEATHEVENT_DEATH_TA
		if object.timer < 288
			music.volume -= 2
			object.timer += 4
		else
			timeAttack.result = 100000
			player.score = 0
			lampPostID = 0
			recMilliSeconds = 0
			recSeconds = 0
			recMinutes = 0
			stage.milliSeconds = 0
			stage.seconds = 0
			stage.minutes = 0
			engine.state = 8

#platform: USE_ORIGINS
			CallNativeFunction4(NotifyCallback, NOTIFY_STATS_ENEMY, StageStatsUsabilityParam1, StageStatsUsabilityParam2, StageStatsUsabilityParam3)
			StageStatsUsabilityParam1 = 0
			StageStatsUsabilityParam2 = 0
			StageStatsUsabilityParam3 = 0
			StageStatsUsabilityParam4 = 0
			StageStatsUsabilityParam5 = 0
			StatsUsabilityParam1 = 0
			StatsUsabilityParam2 = 0
			StatsUsabilityParam3 = 0
			StatsUsabilityParam4 = 0
			StatsUsabilityParam5 = 0
#endplatform
		end if
		SetScreenFade(0, 0, 0, object.timer)
		break

#platform: USE_ORIGINS
	case DEATHEVENT_WAITFORCALLBACK
		if game.callbackResult >= 0
			object.timer = 0
			object.state = DEATHEVENT_RETRY
		end if
		break
		
	case DEATHEVENT_RETRY
		if object.timer < 288
			music.volume -= 2
			object.timer += 4
		else
			if game.callbackResult == 0
				engine.state = 8
			else
				LoadStage()
			end if
		end if
		
		SetScreenFade(0, 0, 0, object.timer)
		break
		
	case DEATHEVENT_SETSCREENFADE
		SetScreenFade(0, 0, 0, 255)
		break
#endplatform
	end switch
end event


event ObjectDraw
	switch object.state
	case DEATHEVENT_GAMEOVER
		// Draw Solid Black Background with fade-in alpha
		DrawRect(0, 0, screen.xsize, screen.ysize, 0x00, 0x00, 0x00, object.timer)

		object.inkEffect = INK_ALPHA
		object.alpha = object.timer

		// 1. Draw static Trunk (Sprite #3 - Back layer)
		DrawSpriteScreenXY(3, screen.xcenter, screen.ycenter)

		// 2. Draw moving Purple Foliage (Sprite #4 - Middle layer)
		temp1 = screen.xcenter
		temp1 += object.value5
		temp2 = screen.ycenter
		temp2 += object.value6
		DrawSpriteScreenXY(4, temp1, temp2)

		// 3. Draw moving Red Foliage (Sprite #5 - Front layer)
		temp1 = screen.xcenter
		temp1 += object.value7
		temp2 = screen.ycenter
		temp2 += object.value8
		DrawSpriteScreenXY(5, temp1, temp2)
		break

	case DEATHEVENT_TIMEOVER
		DrawSpriteScreenXY(1, object.leftTextPos, 96)
		DrawSpriteScreenXY(2, object.rightTextPos, 96)
		break
		
	end switch
end event


event ObjectStartup
	LoadSpriteSheet("Global/Display.gif")

	// Death Event frames
	SpriteFrame(0, 0, 64, 16, 1, 239) 		// 0 - "Game"
	SpriteFrame(0, 0, 56, 16, 66, 239) 		// 1 - "Time"
	SpriteFrame(0, 0, 64, 16, 123, 239) 	// 2 - "Over"

	LoadSpriteSheet("Man/Man.gif")
	SpriteFrame(-25, 2, 58, 44, 28, 48) 		// 3 - Trunk
	SpriteFrame(-43, -30, 87, 47, 115, 16) 		// 4 - Purple Foliage
	SpriteFrame(-52, -45, 105, 68, 213, 1) 		// 5 - Red Foliage
end event

end event

	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event
